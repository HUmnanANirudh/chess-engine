cmake_minimum_required(VERSION 3.16)
project(ChessEngineGUI)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

find_package(Qt5 COMPONENTS Core Gui Widgets REQUIRED)

# Add the GUI executable
add_executable(chess_engine_gui 
    main_gui.cpp 
    chess_gui.cpp 
    chess.cpp
)

target_link_libraries(chess_engine_gui
    Qt5::Core
    Qt5::Gui
    Qt5::Widgets
)

target_include_directories(chess_engine_gui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})

# Compiler flags
if(MSVC)
    target_compile_options(chess_engine_gui PRIVATE /W4)
else()
    target_compile_options(chess_engine_gui PRIVATE -Wall -Wextra -Wpedantic)
endif()
